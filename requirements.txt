# Runtime dependencies — everything src/ and app.py need.
#
# This is the file Streamlit Community Cloud installs, so it is kept minimal,
# headless, and installable *from wheels* on Python 3.10 through 3.14.
# Streamlit Cloud currently provisions Python 3.14, so every pin below must
# have a cp314 wheel or the build falls back to compiling from source. To also run the
# notebooks, install requirements-dev.txt.
# =============================================================================

# --- Image processing -------------------------------------------------------
# Headless build: identical API minus the GUI functions (imshow / namedWindow /
# waitKey), none of which this project calls. The regular opencv-python wheel
# links against libGL.so.1, which is absent from Streamlit Cloud containers.
# Ships as an abi3 wheel, so this pin works on every Python 3.7+.
opencv-python-headless==4.10.0.84

# --- Numerics ---------------------------------------------------------------
# numpy 1.26.4 and pandas 2.2.2 publish wheels for cp39-cp312 only. On Python
# 3.13 pip finds no wheel, tries to build from source, fails — and because that
# aborts the whole install transaction, NOTHING gets installed and the app dies
# on `import cv2` with a misleading ModuleNotFoundError. So the version is made
# conditional: 3.12 and below get the exact pinned builds used in development,
# 3.13+ gets the earliest release with wheels for it. Both combinations were
# verified to produce byte-identical pipeline results.
numpy==1.26.4 ; python_version < "3.13"
numpy>=2.1,<3 ; python_version >= "3.13"

pandas==2.2.2 ; python_version < "3.13"
pandas>=2.2.3,<3 ; python_version >= "3.13"

# --- Visualization ----------------------------------------------------------
matplotlib==3.9.2 ; python_version < "3.13"
matplotlib>=3.10.6 ; python_version >= "3.13"    # 3.10.6 is the first with cp314 wheels

# --- Configuration ----------------------------------------------------------
PyYAML==6.0.2 ; python_version < "3.13"
PyYAML>=6.0.3 ; python_version >= "3.13"         # 6.0.3 is the first with cp314 wheels

# --- Interactive app --------------------------------------------------------
streamlit==1.60.0                  # pure-python wheel; version-independent
