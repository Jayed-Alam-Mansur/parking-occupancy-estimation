# Automatic Parking Occupancy Estimation
# Classical Image Processing Project
#
# Runtime dependencies for src/ and app.py. This is the file Streamlit
# Community Cloud installs, so it is kept minimal, headless, and installable
# from wheels on Python 3.10 through 3.14. Streamlit Cloud currently provisions
# Python 3.14, so every pin below must have a cp314 wheel, otherwise the build
# falls back to compiling from source.
#
# To also run the notebooks, install requirements-dev.txt.

# Image processing.
# The headless build has the same API as opencv-python without the GUI
# functions (imshow, namedWindow, waitKey), none of which this project calls.
# The standard wheel links against libGL.so.1, which is not present in the
# deployment container. The headless wheel is abi3, so this pin works on any
# Python 3.7 or later.
opencv-python-headless==4.10.0.84

# Numerics.
# numpy 1.26.4 and pandas 2.2.2 publish wheels for cp39 to cp312 only. On
# Python 3.13 and later there is no wheel, so the installer attempts a source
# build and fails, which aborts the whole install and leaves no packages
# installed. The version is therefore made conditional: 3.12 and below use the
# pinned development versions, 3.13 and later use the earliest release with a
# published wheel. Both combinations were verified to produce identical results.
numpy==1.26.4 ; python_version < "3.13"
numpy>=2.1,<3 ; python_version >= "3.13"

pandas==2.2.2 ; python_version < "3.13"
pandas>=2.2.3,<3 ; python_version >= "3.13"

# Visualization. 3.10.6 is the first release with a cp314 wheel.
matplotlib==3.9.2 ; python_version < "3.13"
matplotlib>=3.10.6 ; python_version >= "3.13"

# Configuration. 6.0.3 is the first release with a cp314 wheel.
PyYAML==6.0.2 ; python_version < "3.13"
PyYAML>=6.0.3 ; python_version >= "3.13"

# Streamlit application. Pure-python wheel, so the pin is version-independent.
streamlit==1.60.0
